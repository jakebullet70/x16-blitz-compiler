REM === CONSTANTS ===
VIEW.TOP% = 0
VIEW.BOTTOM% = 15
VIEW.LEFT% = 0
VIEW.RIGHT% = 30

BASE% = 8

REM COLORS / CODES
CS% = 6 : CG% = 13
TS% = 32 : TG% = 46 : TT% = 35 : TR% = 94
REM HUD DYNAMIC VALUE COLOR + RUNWAY MARKER TILE/FOREGROUND + KEY COLOR
HC% = 14 : RMK.T% = 64 : RMK.FG% = 7 : KC% = 12
REM MOON USES FOUR 2X2 PETSCII/DISPLAY CODE CELLS
REM TL=$55 TR%=$49 BL=$4A BR%=$4B
MOON.C%=1:MOON.T1%=85:MOON.T2%=73:MOON.T3%=74:MOON.T4%=75
REM UNIFORM PERIOD GRASS FOR SPEED AND CLASSIC LOOK

REM === FLIGHT / GAME STATE ===
PITCH% = 0 : ROLL% = 0
PITCH.PREV% = 999 : ROLL.PREV% = 999

ALT% = 2000
SPD% = 128
FUEL% = 360
THR% = 9
GEAR% = 0
FLP% = 0
RUD% = 0
VSI% = 0
TICK% = 0
SIM.CHANGE% = 0
STALL% = 0
MODE% = 0
REM MODE% 0=FLIGHT 1=LANDED 2=CRASH 3=MISSED 4=EARLY
CRACKED% = 0
BURN% = 0
FUEL.BURN% = 1
LOG.ON% = 1 : LOG.CLOSED% = 1 : SIMTIME% = 0

REM RANDOM WIND AND LANDING DIRECTION
WIND.DIR% = INT(RND(1)*8)
WIND.SPD% = INT(RND(1)*10)
RW.BEAR% = 0
IF WIND.DIR%<=3 THEN RW.BEAR%=180

REM HEADING / TURN% STATE
REM START IN A LOOSE APPROACH CONE, NOT A FULL SEARCH MODE%
HDG% = RW.BEAR% + INT(RND(1)*11)-5
IF HDG%<0 THEN HDG%=HDG%+360
IF HDG%>359 THEN HDG%=HDG%-360
HDG.PREV% = 999
TURN% = 0
TURNSTEP% = 15

DIM OH%(30)
REM V63 PHASE 1: INTEGER ARRAYS FOR HORIZON, OBJECT, AND CRACK STATE.

REM === WORLD OBJECTS ===
REM BB% = WORLD BEARING, BH% = HEIGHT, BW% = WIDTH
REM OS%/OB%/OT%/OV% STORE PREVIOUS SCREEN X, BOTTOM, TOP, VISIBLE
NB% = 3
DIM BB%(3),BH%(3),BW%(3),BC%(3),BD%(3)
DIM OS%(3),OB%(3),OT%(3),OV%(3)
DIM SN%(3),BN%(3),TN%(3),VN%(3)
DIM OWID%(3),OHIG%(3),NWID%(3),NHIG%(3)
REM V71: LAKE MASK IS GENERATED ONCE PER APPROACH, THEN ONLY READ WHILE DRAWING.
DIM LK%(15)

REM CRASH SCREEN CRACK DIRECTIONS / TILES
REM CRACK COUNT / LENGTH ARE BASED ON IMPACT VSI.
REM THESE TILE CODES ARE PETSCII-LIKE LINE GLYPHS.
REM IF YOUR FONT MAP DIFFERS, TWEAK CRH%,CRV%,CRD1%,CRD2%,CRJ.
DIM CDX%(8),CDY%(8),CTK%(8)
CRH%=64:CRV%=93:CRD1%=77:CRD2%=78:CRJ%=91
CDX%(1)=-1:CDY%(1)=0:CTK%(1)=CRH%
CDX%(2)=1:CDY%(2)=0:CTK%(2)=CRH%
CDX%(3)=0:CDY%(3)=-1:CTK%(3)=CRV%
CDX%(4)=0:CDY%(4)=1:CTK%(4)=CRV%
CDX%(5)=-1:CDY%(5)=-1:CTK%(5)=CRD1%
CDX%(6)=1:CDY%(6)=1:CTK%(6)=CRD1%
CDX%(7)=1:CDY%(7)=-1:CTK%(7)=CRD2%
CDX%(8)=-1:CDY%(8)=1:CTK%(8)=CRD2%

REM BUILDINGS START AS SIDE SCENERY DURING APPROACH.
REM THEY ARE OFFSET FROM THE RUNWAY BEARING, NOT IN FRONT OF% IT.
REM BUILDINGS ARE FIXED TO RUNWAY SIDE: CYAN WEST, MAGENTA EAST.
REM NORTH APPROACH: CYAN LEFT/CLOSER, MAGENTA RIGHT/FARTHER.
REM SOUTH APPROACH: SIDES REVERSE, AND MAGENTA BECOMES CLOSER.
IF RW.BEAR%=0 THEN BB%(1)=336:BD%(1)=-20
IF RW.BEAR%=180 THEN BB%(1)=204:BD%(1)=-50
BH%(1)=3 : BW%(1)=2 : BC%(1)=3

IF RW.BEAR%=0 THEN BB%(2)=26:BD%(2)=-50
IF RW.BEAR%=180 THEN BB%(2)=154:BD%(2)=-20
BH%(2)=6 : BW%(2)=3 : BC%(2)=4

REM LAKE LANDMARK: BEHIND START, BIASED WEST OF% RUNWAY COURSE.
REM IT IS DRAWN AS A LOW DARK-BLUE GROUND PATCH, NOT A COLLISION OBJECT.
BB%(3)=RW.BEAR%+200
IF BB%(3)>359 THEN BB%(3)=BB%(3)-360
BH%(3)=3 : BW%(3)=5 : BC%(3)=6 : BD%(3)=0

FOR I=1 TO NB%
  OS%(I)=0 : OB%(I)=0 : OT%(I)=0 : OV%(I)=0
  SN%(I)=0 : BN%(I)=0 : TN%(I)=0 : VN%(I)=0
  OWID%(I)=BW%(I) : OHIG%(I)=BH%(I)
  NWID%(I)=BW%(I) : NHIG%(I)=BH%(I)
NEXT I
GOSUB MAKELAKE

REM === RUNWAY ===
REM RW.BEAR% IS APPROACH / RUNWAY WORLD BEARING.
REM RW.DIST% IS A SIMPLE FAKE APPROACH DISTANCE TO NEAR EDGE.
REM RUNWAY OCCUPIES ABOUT 0 TO -40 DIST UNITS.
REM RWT%/RWC% ARE RUNWAY TILE AND COLOR.
REM START FAR ENOUGH OUT TO CORRECT, BUT CLOSE ENOUGH TO SEE APPROACH
RW.DIST% = 55 + INT(RND(1)*35)
RW.DIST.PREV% = 999
VDIST.PREV% = 999
RW.MAXW% = 19
RWT% = 160 : RWC% = 12
RV% = 0
RW.BLOCK% = 0
RX1% = 0 : RX2% = 0 : RY1% = 0 : RY2% = 0
ORV% = 0 : ORX1% = 0 : ORX2% = 0 : ORY1% = 0 : ORY2% = 0
OMV%=0:OMX%=0:OMY%=0:MOON.BEAR%=0

REM === INIT DRAW BACKGROUND ===
FOR X = VIEW.LEFT% TO VIEW.RIGHT%
  OF% = -(X-CENTER%)*ROLL%/10
  HR% = INT(BASE%+PITCH%+OF%)
  IF HR%<VIEW.TOP% THEN HR%=VIEW.TOP%
  IF HR%>VIEW.BOTTOM% THEN HR%=VIEW.BOTTOM%+1
  OH%(X)=HR%

  FOR Y = VIEW.TOP% TO VIEW.BOTTOM%
    IF Y<HR% THEN TILE X,Y,TS%,CS%
    IF Y>=HR% THEN TILE X,Y,TG%,CG%
  NEXT Y
NEXT X

GOSUB DRAWHUD
GOSUB DRAWHDG
GOSUB DRAWMOON
GOSUB DRAWRUNWAY
GOSUB SETRUNWAYBLOCK
GOSUB DRAWOBJECTS

PITCH.PREV%=PITCH% : ROLL.PREV%=ROLL% : HDG.PREV%=HDG% : RW.DIST.PREV%=RW.DIST% : VDIST.PREV%=INT(RW.DIST%/4)
GOSUB INITLOG
K$=" ":GOSUB LOGCMD

REM === MAIN LOOP ===
MAIN:
SIM.CHANGE%=0
GET K$
IF K$="" THEN GOTO SKIPKEYS
GOSUB LOGCMD
KEYC%=ASC(K$)

REM V74: EARLY-EXIT KEY DISPATCH; ONLY ONE KEY CAN MATCH PER GET.
IF KEYC%>=48 THEN IF KEYC%<=57 THEN THR%=KEYC%-48:SIM.CHANGE%=1:GOTO SKIPKEYS
IF K$="S" THEN PITCH%=PITCH%+1:GOTO SKIPKEYS
IF K$="W" THEN PITCH%=PITCH%-1:GOTO SKIPKEYS
IF K$="A" THEN ROLL%=ROLL%-1:GOTO SKIPKEYS
IF K$="D" THEN ROLL%=ROLL%+1:GOTO SKIPKEYS
IF K$="P" THEN GOSUB PAUSE:GOTO SKIPKEYS
IF K$="G" THEN GOSUB TOGGLEGEAR:GOTO SKIPKEYS
IF K$="F" THEN GOSUB CYCLEFLAPS:GOTO SKIPKEYS
IF K$="U" THEN GOSUB TOGGLEFUELBURN:GOTO SKIPKEYS
IF K$="J" THEN RUD%=RUD%-1:SIM.CHANGE%=1:GOTO SKIPKEYS
IF K$="L" THEN RUD%=RUD%+1:SIM.CHANGE%=1:GOTO SKIPKEYS
IF K$="K" THEN RUD%=0:SIM.CHANGE%=1:GOTO SKIPKEYS

SKIPKEYS:
REM V68: EMPTY FUEL% MEANS ENGINE OFF; IGNORE THROTTLE-UP KEYS.
IF FUEL%<=0 THEN THR%=0:SIM.CHANGE%=1

IF PITCH%<-6 THEN PITCH%=-6
IF PITCH%>6 THEN PITCH%=6
IF ROLL%<-10 THEN ROLL%=-10
IF ROLL%>10 THEN ROLL%=10
IF RUD%<-9 THEN RUD%=-9
IF RUD%>9 THEN RUD%=9

REM === BANK CAUSES HEADING CHANGE ===
REM NEGATIVE ROLL% = LEFT BANK.
REM A LEFT BANK SHOULD MAKE THE LEFT SIDE OF% THE HORIZON DROP.
REM LEFT BANK COUNTS DOWN: 360,359,358...
REM RIGHT BANK COUNTS UP: 360,001,002...
REM RUDDER IS A SMALL HEADING NUDGE.
TURN% = TURN% + ROLL% + INT(RUD%/2)

IF TURN% >= TURNSTEP% THEN HDG%=HDG%+1 : TURN%=TURN%-TURNSTEP%
IF TURN% <= -TURNSTEP% THEN HDG%=HDG%-1 : TURN%=TURN%+TURNSTEP%

IF HDG%<0 THEN HDG%=HDG%+360
IF HDG%>359 THEN HDG%=HDG%-360

REM === SLOW GAME CLOCK ===
TICK%=TICK%+1
IF TICK%>=12 THEN TICK%=0:GOSUB SIMUPDATE
IF MODE%=1 THEN GOSUB LOGRESULT:GOSUB DRAWDYNHUD:GOTO LANDEDLOOP
IF MODE%=2 THEN GOSUB LOGRESULT:GOSUB CRACKSCREEN:GOSUB DRAWDYNHUD:GOTO CRASHLOOP
IF MODE%=3 THEN GOSUB LOGRESULT:GOSUB DRAWDYNHUD:GOTO MISSEDLOOP
IF MODE%=4 THEN GOSUB LOGRESULT:GOSUB DRAWDYNHUD:GOTO EARLYLOOP

REM V42 PERFORMANCE: ONLY REDRAW RUNWAY/WORLD WHEN A COARSER
REM VISIBLE DISTANCE BUCKET CHANGES. HUD DIST STILL UPDATES EVERY SIM TICK.
VDIST%=INT(RW.DIST%/6)
IF PITCH%=PITCH.PREV% AND ROLL%=ROLL.PREV% AND HDG%=HDG.PREV% AND VDIST%=VDIST.PREV% THEN IF SIM.CHANGE%=0 THEN GOTO MAIN
IF PITCH%=PITCH.PREV% AND ROLL%=ROLL.PREV% AND HDG%=HDG.PREV% AND VDIST%=VDIST.PREV% THEN IF SIM.CHANGE%=1 THEN GOSUB DRAWDYNHUD:GOTO MAIN

GOSUB SETRUNWAYBLOCK
GOSUB PREPOBJECTS

REM CONSERVATIVE DRAW ORDER TO AVOID RUNWAY/BUILDING ARTIFACTS.
REM OLD OBJECTS AND OLD RUNWAY ARE FULLY RESTORED BEFORE THE HORIZON DELTA,
REM THEN THE NEW RUNWAY AND OBJECTS ARE DRAWN BACK IN ORDER.
GOSUB SMARTERASEOBJECTS
GOSUB ERASERUNWAY
GOSUB ERASEMOON

IF PITCH%<>PITCH.PREV% OR ROLL%<>ROLL.PREV% THEN GOSUB DELTA

GOSUB DRAWMOON
GOSUB DRAWRUNWAY
GOSUB DRAWPENDINGOBJECTS

IF HDG%<>HDG.PREV% THEN GOSUB DRAWHDG
IF SIM.CHANGE%=1 THEN GOSUB DRAWDYNHUD

PITCH.PREV%=PITCH% : ROLL.PREV%=ROLL% : HDG.PREV%=HDG% : RW.DIST.PREV%=RW.DIST% : VDIST.PREV%=VDIST%
GOTO MAIN

REM === FLIGHT LOG ===
INITLOG:
SIMTIME%=0
IF LOG.ON%=0 THEN LOG.CLOSED%=1:RETURN
OPEN 1,8,1,"@:FLIGHT.LOG,S,W"
LOG.CLOSED%=0
PRINT#1,"X16 FLIGHT SIM LOG"
PRINT#1,"T KEY SPD ALT VSI THR FLP GEAR DIST"
PRINT#1,"--- NEW APPROACH ---"
GOSUB LOGINIT
RETURN

LOGINIT:
IF LOG.ON%=0 THEN RETURN
IF LOG.CLOSED%=1 THEN RETURN
PRINT#1,"INIT ALT=";INT(ALT%);" SPD=";INT(SPD%);" FUEL=";INT(FUEL%);" VSI=";INT(VSI%)
PRINT#1,"INIT THR=";THR%;" FLP=";FLP%;" GEAR=";GEAR%;" RUD=";RUD%;" PITCH=";PITCH%;" ROLL=";ROLL%;" FUEL.BURN=";FUEL.BURN%
PRINT#1,"INIT HDG=";INT(HDG%);" RW.BEAR=";INT(RW.BEAR%);" RW.DIST=";INT(RW.DIST%)
PRINT#1,"INIT WIND.DIR=";WIND.DIR%;" WIND.SPD=";WIND.SPD%
PRINT#1,"INIT BB1=";BB%(1);" BB2=";BB%(2);" BB3=";BB%(3)
PRINT#1,"INIT BD1=";BD%(1);" BD2=";BD%(2)

T$=STR$(SIMTIME%):T$=MID$(T$,2)
IF SIMTIME%<10000 THEN T$="0"+T$
IF SIMTIME%<1000 THEN T$="0"+T$
IF SIMTIME%<100 THEN T$="0"+T$
IF SIMTIME%<10 THEN T$="0"+T$
RETURN

LOGCMD:
IF LOG.ON%=0 THEN RETURN
IF LOG.CLOSED%=1 THEN RETURN
GOSUB FORMATTIME
LK$=K$
IF K$=" " THEN LK$="SPACE"
PRINT#1,"T=";T$;" K=";LK$;" SPD=";INT(SPD%);" ALT=";INT(ALT%);" VSI=";INT(VSI%);" THR=";THR%;" FLP=";FLP%;" GEAR=";GEAR%;" DIST=";INT(RW.DIST%)
RETURN

LOGRESULT:
IF LOG.ON%=0 THEN RETURN
IF LOG.CLOSED%=1 THEN RETURN
GOSUB FORMATTIME
IF MODE%=1 THEN PRINT#1,"T=";T$;" RESULT=LANDED SPD=";INT(SPD%);" ALT=";INT(ALT%);" VSI=";INT(VSI%);" DIST=";INT(RW.DIST%)
IF MODE%=2 THEN PRINT#1,"T=";T$;" RESULT=CRASH SPD=";INT(SPD%);" ALT=";INT(ALT%);" VSI=";INT(VSI%);" DIST=";INT(RW.DIST%)
IF MODE%=3 THEN PRINT#1,"T=";T$;" RESULT=MISSED SPD=";INT(SPD%);" ALT=";INT(ALT%);" VSI=";INT(VSI%);" DIST=";INT(RW.DIST%)
IF MODE%=4 THEN PRINT#1,"T=";T$;" RESULT=EARLY SPD=";INT(SPD%);" ALT=";INT(ALT%);" VSI=";INT(VSI%);" DIST=";INT(RW.DIST%)
GOSUB CLOSELOG
RETURN

IF LOG.CLOSED%=1 THEN RETURN

IF LOG.ON%=0 THEN RETURN
IF LOG.CLOSED%=0 THEN RETURN
REM V82: START EACH NEW APPROACH WITH A FRESH LOG FILE.
REM END-STATES CLOSE THE FILE SO IT CAN BE VIEWED OR COPIED.
OPEN 1,8,1,"@:FLIGHT.LOG,S,W"

PRINT#1,"X16 FLIGHT SIM LOG"

GOSUB CLOSELOG

END

REM === SIM UPDATE ===
REM SIMPLE GAME SCAFFOLDING, NOT REAL PHYSICS YET.
REM V26 TUNING: FASTER SIM TICK%, QUICKER SPD% RESPONSE, BETTER LEVEL CLOSURE.
SIMUPDATE:
SIMTIME%=SIMTIME%+1
REM V68: IF TANK IS EMPTY, ENGINE IS OFF BEFORE PHYSICS RUNS.
IF FUEL%<=0 THEN THR%=0:BURN%=0
REM SPEED TARGET: THROTTLE ADDS SPEED, FLAPS ADD DRAG.
REM V41: THR% 9 SUSTAINS ABOUT 128 IN LEVEL CLEAN FLIGHT.
REM THR% 3-4 WITH FLAPS IS LANDING RANGE; THR% 1-2 IS NEAR STALL.
SPD.TAR% = 47 + THR%*9 - FLP%*8 - GEAR%*8
IF THR%=0 THEN SPD.TAR%=28 - FLP%*7 - GEAR%*5
IF SPD.TAR%<18 THEN SPD.TAR%=18
IF SPD%<SPD.TAR% THEN SPD%=SPD%+2
IF SPD%>SPD.TAR% THEN SPD%=SPD%-2
IF SPD%<SPD.TAR% THEN IF SPD.TAR%-SPD%<2 THEN SPD%=SPD.TAR%
IF SPD%>SPD.TAR% THEN IF SPD%-SPD.TAR%<2 THEN SPD%=SPD.TAR%
IF SPD%<0 THEN SPD%=0

REM SIMPLE CLIMB MODEL:
REM V43: THROTTLE BELOW 6 NOW CREATES CLEAR SINK.
REM THR% 9 SUSTAINS SPEED, BUT DOES NOT MAGICALLY CLIMB WITHOUT PITCH.
REM GEAR% DOWN ADDS DRAG: LOWER SPEED TARGET AND MORE DESCENT.
IF PITCH%<0 THEN VSI% = PITCH%*650 + (THR%-6)*15 - FLP%*170 - GEAR%*120
IF PITCH%>=0 THEN VSI% = PITCH%*330 + (THR%-6)*35 - FLP%*260 - GEAR%*120
IF THR%<6 THEN VSI%=VSI%-(6-THR%)*140
IF THR%<=2 THEN VSI%=VSI%-250
IF THR%=0 THEN IF VSI%> -1000 THEN VSI%=-1000

REM V42: DISPLAYED VSI% IS LESS EXAGGERATED, BUT ALT% RESPONDS MORE.
REM THIS MAKES THE INSTRUMENT FEEL CLOSER TO WHAT THE AIRCRAFT DOES.
VSI%=INT(VSI%*2/3)

REM NEAR-GROUND PITCH% AUTHORITY: A FLARE SHOULD MATTER.
IF ALT%<300 THEN VSI%=VSI%+PITCH%*120

REM BASIC STALL% MODEL:
REM FLAPS LOWER STALL% SPEED SLIGHTLY. STALL% FORCES A HARD DESCENT.
STALL%=0
STSPD% = 55 - FLP%*5
IF SPD%<STSPD% AND ALT%>0 THEN STALL%=1
IF STALL%=1 THEN VSI%=-1800

IF VSI%<-2800 THEN VSI%=-2800
IF VSI%>1600 THEN VSI%=1600
ALT% = ALT% + INT(VSI%/25)
IF ALT%<0 THEN ALT%=0
IF ALT%>99999 THEN ALT%=99999

IF FUEL.BURN%=1 THEN IF THR%>0 THEN BURN%=BURN%+THR%+GEAR%+FLP%
IF FUEL.BURN%=1 THEN IF BURN%>=10 THEN FUEL%=FUEL%-1:BURN%=BURN%-10
IF FUEL%<0 THEN FUEL%=0
IF FUEL%=0 THEN THR%=0:BURN%=0

REM RUNWAY DISTANCE USES GROUND SPEED, NOT JUST AIRSPEED.
REM V37: DO NOT LET ORDINARY DESCENT ZERO OUT FORWARD PROGRESS.
BR% = RW.BEAR%-HDG%
IF BR%>180 THEN BR%=BR%-360
IF BR%<-179 THEN BR%=BR%+360
ABR%=ABS(BR%)
GS% = SPD% - INT(ABS(VSI%)/90)
IF GS%<0 THEN GS%=0

REM V55: 8-DIRECTION WIND. WIND.DIR% IS WHERE WIND COMES FROM.
WD.BEAR% = WIND.DIR%*45
WREL% = WD.BEAR%-HDG%
IF WREL%>180 THEN WREL%=WREL%-360
IF WREL%<-179 THEN WREL%=WREL%+360
IF ABS(WREL%)<45 THEN GS%=GS%-WIND.SPD%
IF ABS(WREL%)>135 THEN GS%=GS%+WIND.SPD%
IF GS%<0 THEN GS%=0
IF ABS(WREL%)>=45 THEN IF ABS(WREL%)<=135 THEN TURN%=TURN%+INT(WIND.SPD%/2)

REM V39: KEEP% +60 WIGGLE ROOM, THEN CLOSE FASTER ON FINAL.
ADV%=INT(GS%/45)
IF RW.DIST%>25 THEN ADV%=INT(GS%/60)
IF RW.DIST%<=25 THEN ADV%=INT(GS%/38)
IF RW.DIST%<=0 THEN ADV%=INT(GS%/32)
IF SPD%>45 THEN IF ADV%<1 THEN ADV%=1
IF THR%=0 THEN IF SPD%<55 THEN ADV%=0
IF ABR%<45 THEN RW.DIST%=RW.DIST%-ADV%
IF ABR%>=45 THEN RW.DIST%=RW.DIST%+1
IF RW.DIST%<-60 THEN RW.DIST%=-60
IF RW.DIST%>120 THEN RW.DIST%=120

GOSUB CHECKSTATUS
SIM.CHANGE%=1
RETURN

REM === LANDING / CRASH DETECTION ===
CHECKSTATUS:
REM BUILDING COLLISION: LOW ALT%, SAME BEARING, SAME RUNWAY DISTANCE.
IF ALT%>0 THEN GOSUB CHECKBUILDINGS
IF MODE%=2 THEN RETURN
IF ALT%>0 THEN IF RW.DIST%<=-60 THEN MODE%=3:RETURN
IF ALT%>0 THEN RETURN
BR2% = RW.BEAR%-HDG%
IF BR2%>180 THEN BR2%=BR2%-360
IF BR2%<-179 THEN BR2%=BR2%+360
ONRW%=0
IF ABS(BR2%)<8 THEN IF RW.DIST%<=5 THEN IF RW.DIST%>=-60 THEN ONRW%=1
GOODLAND%=0
IF GEAR%=1 THEN IF SPD%<105 THEN IF ABS(ROLL%)<4 THEN IF VSI%>-1200 THEN GOODLAND%=1
IF ONRW%=1 THEN IF GOODLAND%=1 THEN MODE%=1:RETURN
IF ONRW%=1 THEN MODE%=2:RETURN
IF GOODLAND%=1 THEN IF RW.DIST%>5 THEN MODE%=4:RETURN
IF GOODLAND%=1 THEN IF RW.DIST%<-60 THEN MODE%=3:RETURN
MODE%=2
RETURN

CHECKBUILDINGS:
FOR I=1 TO 2
  BR2% = BB%(I)-HDG%
  IF BR2%>180 THEN BR2%=BR2%-360
  IF BR2%<-179 THEN BR2%=BR2%+360
  DD% = RW.DIST%-BD%(I)
  IF ALT%<400 THEN IF ABS(BR2%)<5 THEN IF ABS(DD%)<7 THEN MODE%=2:RETURN
NEXT I
RETURN

REM === CRASH CRACK SCREEN ===
REM DRAWS STRUCTURED PETSCII-LIKE CRACKS INSIDE THE VIEW WINDOW ONLY.
REM ONE CENTRAL IMPACT PLUS NEARBY CRACKS, WITH MIXED LINE GLYPHS.
REM THIS IS ONLY CALLED ONCE PER CRASH STATE.
CRACKSCREEN:
IF CRACKED%=1 THEN RETURN
CRACKED%=1

REM FIXED TWO-CLUSTER CRACK PATTERN.
REM THIS READS MORE LIKE WINDSHIELD DAMAGE AND LESS LIKE A GLITCH.
CRN%=2:CRM%=5:CRB%=5

ICX%=CENTER%
ICY%=BASE%+INT(RND(1)*5)-2
IF ICX%<VIEW.LEFT%+2 THEN ICX%=VIEW.LEFT%+2
IF ICX%>VIEW.RIGHT%-2 THEN ICX%=VIEW.RIGHT%-2
IF ICY%<VIEW.TOP%+2 THEN ICY%=VIEW.TOP%+2
IF ICY%>VIEW.BOTTOM%-2 THEN ICY%=VIEW.BOTTOM%-2

FOR CP=1 TO CRN%
  IF CP=1 THEN CX%=ICX%-6
  IF CP=2 THEN CX%=ICX%+6
  CY%=ICY%+INT(RND(1)*3)-1
  IF CX%<VIEW.LEFT%+2 THEN CX%=VIEW.LEFT%+2
  IF CX%>VIEW.RIGHT%-2 THEN CX%=VIEW.RIGHT%-2
  IF CY%<VIEW.TOP%+2 THEN CY%=VIEW.TOP%+2
  IF CY%>VIEW.BOTTOM%-2 THEN CY%=VIEW.BOTTOM%-2
  TILE CX%,CY%,CRJ%,1
  FOR DIR=1 TO 8
    IF INT(RND(1)*2)=0 THEN GOTO CRACKNEXTDIR
    CLEN%=2+INT(RND(1)*CRM%)
    XX%=CX%:YY%=CY%
    FOR STP=1 TO CLEN%
      XX%=XX%+CDX%(DIR)
      YY%=YY%+CDY%(DIR)
      IF XX%<VIEW.LEFT% THEN GOTO CRACKRAYDONE
      IF XX%>VIEW.RIGHT% THEN GOTO CRACKRAYDONE
      IF YY%<VIEW.TOP% THEN GOTO CRACKRAYDONE
      IF YY%>VIEW.BOTTOM% THEN GOTO CRACKRAYDONE
      CRT%=CTK%(DIR)
      IF INT(RND(1)*16)=0 THEN CRT%=CRJ%
      TILE XX%,YY%,CRT%,1
      IF STP>1 THEN IF INT(RND(1)*CRB%)=0 THEN GOSUB CRACKBRANCH
      IF INT(RND(1)*8)=0 THEN GOTO CRACKRAYDONE
    NEXT STP
CRACKRAYDONE:
CRACKNEXTDIR:
  NEXT DIR
NEXT CP
RETURN

CRACKBRANCH:
BDIR%=1+INT(RND(1)*8)
BLEN%=1+INT(RND(1)*3)
BXX%=XX%:BYY%=YY%
FOR BSTP=1 TO BLEN%
  BXX%=BXX%+CDX%(BDIR%)
  BYY%=BYY%+CDY%(BDIR%)
  IF BXX%<VIEW.LEFT% THEN RETURN
  IF BXX%>VIEW.RIGHT% THEN RETURN
  IF BYY%<VIEW.TOP% THEN RETURN
  IF BYY%>VIEW.BOTTOM% THEN RETURN
  CRT%=CTK%(BDIR%)
  IF INT(RND(1)*8)=0 THEN CRT%=CRJ%
  TILE BXX%,BYY%,CRT%,1
NEXT BSTP
RETURN

REM === LANDED / CRASH WAIT LOOPS ===
LANDEDLOOP:
SPD%=0:THR%=0:ALT%=0:VSI%=0
GOSUB DRAWDYNHUD
GET K$
IF K$=" " THEN GOSUB RESETAPPROACH:GOTO MAIN
IF K$="R" THEN GOSUB RESETAPPROACH:GOTO MAIN
IF K$="X" THEN GOTO EXITGAME
GOTO LANDEDLOOP

CRASHLOOP:
SPD%=0:THR%=0:ALT%=0:VSI%=0
GOSUB DRAWDYNHUD
GET K$
IF K$=" " THEN GOSUB RESETAPPROACH:GOTO MAIN
IF K$="R" THEN GOSUB RESETAPPROACH:GOTO MAIN
IF K$="X" THEN GOTO EXITGAME
GOTO CRASHLOOP

MISSEDLOOP:
SPD%=0:THR%=0:VSI%=0
GOSUB DRAWDYNHUD
GET K$
IF K$=" " THEN GOSUB RESETAPPROACH:GOTO MAIN
IF K$="R" THEN GOSUB RESETAPPROACH:GOTO MAIN
IF K$="X" THEN GOTO EXITGAME
GOTO MISSEDLOOP

RESETAPPROACH:

ALT%=2000:SPD%=128:FUEL%=360:THR%=9:GEAR%=0:FLP%=0
VSI%=0:TICK%=0:SIM.CHANGE%=1:STALL%=0:MODE%=0:CRACKED%=0:BURN%=0:FUEL.BURN%=1
WIND.DIR%=INT(RND(1)*8)
WIND.SPD%=INT(RND(1)*10)
RW.BEAR%=0
IF WIND.DIR%<=3 THEN RW.BEAR%=180
HDG%=RW.BEAR%+INT(RND(1)*11)-5
IF HDG%<0 THEN HDG%=HDG%+360
IF HDG%>359 THEN HDG%=HDG%-360
TURN%=0
IF RW.BEAR%=0 THEN BB%(1)=336:BD%(1)=-20
IF RW.BEAR%=180 THEN BB%(1)=204:BD%(1)=-50
IF RW.BEAR%=0 THEN BB%(2)=26:BD%(2)=-50
IF RW.BEAR%=180 THEN BB%(2)=154:BD%(2)=-20
BB%(3)=RW.BEAR%+200
IF BB%(3)>359 THEN BB%(3)=BB%(3)-360
BH%(3)=3:BW%(3)=5
GOSUB MAKELAKE
RW.DIST%=55+INT(RND(1)*35)
RW.DIST.PREV%=999:VDIST.PREV%=999
RV%=0:ORV%=0:OMV%=0
FOR I=1 TO NB%
  OV%(I)=0:VN%(I)=0
NEXT I
FOR X=VIEW.LEFT% TO VIEW.RIGHT%
  OF%=-(X-CENTER%)*ROLL%/10
  HR%=INT(BASE%+PITCH%+OF%)
  IF HR%<VIEW.TOP% THEN HR%=VIEW.TOP%

SIMTIME%=0

REM === DELTA HORIZON ===
REM HORIZON ROWS ARE FORCED TO INTEGER ROWS.
REM THIS PREVENTS FRACTIONAL FOR/NEXT TILE GAPS.
DELTA:
FOR X = VIEW.LEFT% TO VIEW.RIGHT%
  HR.PREV%=OH%(X)

  OF%=-(X-CENTER%)*ROLL%/10
  HR.NEXT%=INT(BASE%+PITCH%+OF%)

  CLIPPED%=0
  IF HR.NEXT%<VIEW.TOP% THEN HR.NEXT%=VIEW.TOP%:CLIPPED%=1
  IF HR.NEXT%>VIEW.BOTTOM% THEN HR.NEXT%=VIEW.BOTTOM%+1:CLIPPED%=1

  IF CLIPPED%=1 THEN GOSUB FULLCOLUMN
  IF CLIPPED%=0 THEN IF HR.NEXT%>HR.PREV% THEN GOSUB SKYBAND
  IF CLIPPED%=0 THEN IF HR.NEXT%<HR.PREV% THEN GOSUB GROUNDBAND

  OH%(X)=HR.NEXT%
NEXT X
RETURN

FULLCOLUMN:
FOR Y=VIEW.TOP% TO VIEW.BOTTOM%
  IF Y<HR.NEXT% THEN TILE X,Y,TS%,CS%
  IF Y>=HR.NEXT% THEN TILE X,Y,TG%,CG%
NEXT
RETURN

SKYBAND:
FOR Y=HR.PREV% TO HR.NEXT%-1
  IF Y>=VIEW.TOP% AND Y<=VIEW.BOTTOM% THEN TILE X,Y,TS%,CS%
NEXT
RETURN

GROUNDBAND:
FOR Y=HR.NEXT% TO HR.PREV%-1
  IF Y>=VIEW.TOP% AND Y<=VIEW.BOTTOM% THEN TILE X,Y,TG%,CG%
NEXT
RETURN

REM === WORLD OBJECT PROJECTION ===
REM INPUT: I
REM OUTPUT: VIS%, SX%, TB%, TTOP%
PROJECT:
BR% = BB%(I)-HDG%
IF BR%>180 THEN BR%=BR%-360
IF BR%<-179 THEN BR%=BR%+360

REM WHEN RUNWAY IS THE PRIMARY VIEW, ONLY HIDE BUILDINGS IN THE CENTER% CORRIDOR.
REM SIDE BUILDINGS REMAIN VISIBLE AS APPROACH SCENERY.
IF RW.BLOCK%=1 THEN IF ABS(BR%)<10 THEN VIS%=0:RETURN

REM 2 DEGREES PER COLUMN, ABOUT 60 DEGREE FIELD OF% VIEW
SX% = INT(CENTER% + BR%/2)

REM SCALE BUILDINGS AS RUNWAY LANDMARKS USING PER-BUILDING DISTANCE.
REM BD% IS RUNWAY-RELATIVE DEPTH; CLOSER/FARTHER SWAPS WITH RUNWAY DIRECTION.
BDIST% = RW.DIST% - BD%(I)
REM BUILDINGS DISAPPEAR AFTER THE AIRCRAFT PASSES THEIR RUNWAY POSITION.
REM LAKE REMAINS START/SCENERY RELATIVE AND DOES NOT USE PASS-BEHIND LOGIC.
IF I<3 THEN IF BDIST%<0 THEN VIS%=0:RETURN
IF I=3 THEN BDIST%=35
IF BDIST%<5 THEN BDIST%=5
IF BDIST%>90 THEN BDIST%=90
SH% = INT(11 - BDIST%/10)
IF SH%<BH%(I) THEN SH%=BH%(I)
IF SH%>9 THEN SH%=9
REM CYAN SIDE BUILDING IS A SHORTER ANGLED-ROOF TOWER.
IF I=1 THEN SH%=INT(SH%/2)+1
IF I=1 THEN IF SH%<BH%(I) THEN SH%=BH%(I)
IF I=1 THEN IF SH%>5 THEN SH%=5
SW% = INT(SH%/3)+1
IF SW%<BW%(I) THEN SW%=BW%(I)
IF SW%>5 THEN SW%=5
IF I=3 THEN SH%=BH%(I):SW%=BW%(I)

IF SX%<VIEW.LEFT% THEN VIS%=0:RETURN
IF SX%+SW%-1>VIEW.RIGHT% THEN VIS%=0:RETURN

VIS%=1
TB% = INT(13 + PITCH% - (SX%-CENTER%)*ROLL%/10)
TTOP% = TB%-SH%+1

IF TB%<VIEW.TOP% THEN VIS%=0:RETURN
IF TTOP%>VIEW.BOTTOM% THEN VIS%=0:RETURN

IF TTOP%<VIEW.TOP% THEN TTOP%=VIEW.TOP%
IF TB%>VIEW.BOTTOM% THEN TB%=VIEW.BOTTOM%
RETURN

REM === DRAW OBJECTS ===
REM DRAWOBJECTS IS USED FOR INITIAL FULL OBJECT DRAW.
DRAWOBJECTS:
FOR I=1 TO NB%
  GOSUB PROJECT

  IF VIS%=0 THEN OV%(I)=0:GOTO DRAWOBJNEXT

  FOR X=SX% TO SX%+SW%-1
    YTOP%=TTOP%
    IF I=1 THEN IF SW%>1 THEN YTOP%=TTOP%+INT((SX%+SW%-1-X)*2/(SW%-1))
    IF YTOP%>TB% THEN YTOP%=TB%
    FOR Y=YTOP% TO TB%
      IF I=3 THEN GOSUB DRAWLAKECELL:GOTO DRAWCELLDONE
      IF Y=YTOP% THEN TILE X,Y,TR%,BC%(I)
      IF Y<>YTOP% THEN TILE X,Y,TT%,BC%(I)
DRAWCELLDONE:
    NEXT Y
  NEXT X

  OS%(I)=SX% : OB%(I)=TB% : OT%(I)=TTOP% : OV%(I)=1
  OWID%(I)=SW% : OHIG%(I)=SH%

DRAWOBJNEXT:
NEXT I
RETURN

REM === PREPARE NEW OBJECT POSITIONS ===
PREPOBJECTS:
FOR I=1 TO NB%
  GOSUB PROJECT
  IF VIS%=0 THEN VN%(I)=0:GOTO PREPOBJNEXT
  SN%(I)=SX% : BN%(I)=TB% : TN%(I)=TTOP% : VN%(I)=1
  NWID%(I)=SW% : NHIG%(I)=SH%
PREPOBJNEXT:
NEXT I
RETURN

REM === SMART ERASE OBJECTS ===
REM ERASE ONLY OLD OBJECT CELLS NOT STILL COVERED BY NEW OBJECT.
REM THIS REDUCES FLICKER DURING SMALL HEADING / BANK CHANGES.
SMARTERASEOBJECTS:
FOR I=1 TO NB%
  IF OV%(I)=0 THEN GOTO SERASEOBJNEXT

  SX%=OS%(I) : TB%=OB%(I) : TTOP%=OT%(I)

  FOR Y=TTOP% TO TB%
    FOR X=SX% TO SX%+OWID%(I)-1
      KEEP%=0
      IF VN%(I)=1 THEN IF X>=SN%(I) AND X<SN%(I)+NWID%(I) THEN TOPN%=TN%(I):IF I=1 THEN IF NWID%(I)>1 THEN TOPN%=TN%(I)+INT((SN%(I)+NWID%(I)-1-X)*2/(NWID%(I)-1))
      IF VN%(I)=1 THEN IF X>=SN%(I) AND X<SN%(I)+NWID%(I) THEN IF Y>=TOPN% AND Y<=BN%(I) THEN KEEP%=1
      IF I=3 THEN IF KEEP%=1 THEN LX%=X-SN%(I):LY%=Y-TN%(I):LI=LY%*5+LX%+1
      IF I=3 THEN IF KEEP%=1 THEN IF LI<1 THEN KEEP%=0
      IF I=3 THEN IF KEEP%=1 THEN IF LI>15 THEN KEEP%=0
      IF I=3 THEN IF KEEP%=1 THEN IF LK%(LI)=0 THEN KEEP%=0
      IF KEEP%=0 THEN IF X>=VIEW.LEFT% AND X<=VIEW.RIGHT% THEN GOSUB RESTORECELL
    NEXT X
  NEXT Y

SERASEOBJNEXT:
NEXT I
RETURN

REM === DRAW PREPARED OBJECTS ===
DRAWPENDINGOBJECTS:
FOR I=1 TO NB%
  IF VN%(I)=0 THEN OV%(I)=0:GOTO DPOBJNEXT

  SX%=SN%(I) : TB%=BN%(I) : TTOP%=TN%(I)

  FOR X=SX% TO SX%+NWID%(I)-1
    YTOP%=TTOP%
    IF I=1 THEN IF NWID%(I)>1 THEN YTOP%=TTOP%+INT((SX%+NWID%(I)-1-X)*2/(NWID%(I)-1))
    IF YTOP%>TB% THEN YTOP%=TB%
    FOR Y=YTOP% TO TB%
      IF I=3 THEN GOSUB DRAWLAKECELL:GOTO DRAWCELLDONE2
      IF Y=YTOP% THEN TILE X,Y,TR%,BC%(I)
      IF Y<>YTOP% THEN TILE X,Y,TT%,BC%(I)
DRAWCELLDONE2:
    NEXT Y
  NEXT X

  OS%(I)=SX% : OB%(I)=TB% : OT%(I)=TTOP% : OV%(I)=1
  OWID%(I)=NWID%(I) : OHIG%(I)=NHIG%(I)

DPOBJNEXT:
NEXT I
RETURN

REM === LAKE SHAPE ===
REM V71: MAKE A SMALL 5X3 IRREGULAR LAKE ONCE PER SCENARIO.
MAKELAKE:
FOR LI=1 TO 15
  LK%(LI)=1
NEXT LI
REM TRIM FIXED CORNERS SO THE LAKE IS NEVER A PLAIN RECTANGLE.
LK%(1)=0:LK%(5)=0:LK%(11)=0:LK%(15)=0
REM RANDOM EDGE BITES.  SHAPE STAYS FIXED UNTIL NEXT APPROACH.
R%=1+INT(RND(1)*4)
IF R%=1 THEN LK%(2)=0
IF R%=2 THEN LK%(4)=0
IF R%=3 THEN LK%(12)=0
IF R%=4 THEN LK%(14)=0
R%=1+INT(RND(1)*4)
IF R%=1 THEN LK%(6)=0
IF R%=2 THEN LK%(10)=0
IF R%=3 THEN LK%(7)=0
IF R%=4 THEN LK%(9)=0
RETURN

DRAWLAKECELL:
LX%=X-SX%
LY%=Y-TTOP%
LI=LY%*5+LX%+1
IF LI<1 THEN RETURN
IF LI>15 THEN RETURN
IF LK%(LI)=0 THEN RETURN
TILE X,Y,160,BC%(I)
RETURN

REM === RUNWAY PRIORITY ===
REM IF THE RUNWAY IS CLOSE AND CENTERED, SUPPRESS BUILDINGS.
REM FAR APPROACH KEEPS BUILDINGS AS LANDMARKS.
SETRUNWAYBLOCK:
RW.BLOCK%=0
RBR% = RW.BEAR%-HDG%
IF RBR%>180 THEN RBR%=RBR%-360
IF RBR%<-179 THEN RBR%=RBR%+360
ARBR%=ABS(RBR%)
REM ONLY SUPPRESS BUILDINGS WHEN RUNWAY IS CLOSE AND DOMINANT.
REM FAR AWAY, KEEP% LANDMARKS VISIBLE EVEN WHEN ALIGNED.
IF RW.DIST%<30 AND ARBR%<25 THEN RW.BLOCK%=1
RETURN

REM === RUNWAY ===
REM SIMPLE GROUND-ONLY TRAPEZOID AT RUNWAY APPROACH BEARING.
REM IT GETS LARGER AS RW.DIST% SHRINKS.
DRAWRUNWAY:
REM DRAW RUNWAY AS A FILLED TRAPEZOID.
REM DIST > 0 IS APPROACH VIEW.
REM DIST <= 0 IS ON-RUNWAY / NEAR-FIELD VIEW.
REM WHEN DIST IS ABOUT -40 TO -60, A FEW ROWS OF% GREEN APPEAR
REM AT THE FAR END AS A VISUAL WARNING BEFORE MISSED.
RV%=0
BR% = RW.BEAR%-HDG%
IF BR%>180 THEN BR%=BR%-360
IF BR%<-179 THEN BR%=BR%+360
IF BR%<-30 THEN RETURN
IF BR%>30 THEN RETURN

RXC% = INT(CENTER% + BR%/2)
IF RXC%<VIEW.LEFT% THEN RETURN
IF RXC%>VIEW.RIGHT% THEN RETURN

REM ON-RUNWAY VIEW IS ITS OWN ROUTINE SO IT CAN REALLY TOUCH
REM THE HORIZON LINE INSTEAD OF% BEING A SHORT APPROACH BLOB.
IF RW.DIST%<=0 THEN GOTO DRAWRWYNEAR

RTOP% = OH%(RXC%)+1
IF RTOP%<VIEW.TOP% THEN RTOP%=VIEW.TOP%
IF RTOP%>VIEW.BOTTOM% THEN RETURN

RD% = RW.DIST%
IF RD%>90 THEN RD%=90

REM APPROACH VIEW: FAR RUNWAY IS A SMALL TRAPEZOID BELOW HORIZON.
RW.H% = INT(7 - RD%/20)
IF RW.H%<2 THEN RW.H%=2
IF RW.H%>7 THEN RW.H%=7
RW.WBASE% = INT(15 - RD%/6)
IF RW.WBASE%<3 THEN RW.WBASE%=3
IF RW.WBASE%>RW.MAXW% THEN RW.WBASE%=RW.MAXW%

REM NEAR-THRESHOLD BLEND: WHEN DIST IS 10..0, RUNWAY GROWS
REM TOWARD THE DASH AND WIDENS TO FULL RUNWAY WIDTH. THIS AVOIDS
REM A SUDDEN POP WHEN CROSSING DIST 000.
IF RW.DIST%<=10 THEN NR%=10-RW.DIST%
IF RW.DIST%<=10 THEN FULLH%=VIEW.BOTTOM%-RTOP%+1
IF RW.DIST%<=10 THEN TH%=INT(FULLH%*NR%/10)
IF RW.DIST%<=10 THEN IF TH%>RW.H% THEN RW.H%=TH%
IF RW.DIST%<=10 THEN RW.WBASE%=15+INT(NR%*4/10)
IF RW.WBASE%>RW.MAXW% THEN RW.WBASE%=RW.MAXW%

RY1% = VIEW.BOTTOM%
RY2% = VIEW.TOP%
RX1% = VIEW.RIGHT%
RX2% = VIEW.LEFT%

FOR N=1 TO RW.H%
  Y = RTOP%+N-1
  IF Y<VIEW.TOP% THEN GOTO RWYAPNEXT
  IF Y>VIEW.BOTTOM% THEN GOTO RWYAPNEXT
  W% = INT(N*RW.WBASE%/RW.H%)
  IF W%<1 THEN W%=1
  IF W%>RW.WBASE% THEN W%=RW.WBASE%
  HW% = INT(W%/2)
  X1% = RXC%-HW%
  X2% = RXC%+HW%
  IF X1%<VIEW.LEFT% THEN X1%=VIEW.LEFT%
  IF X2%>VIEW.RIGHT% THEN X2%=VIEW.RIGHT%

  IF X1%<RX1% THEN RX1%=X1%
  IF X2%>RX2% THEN RX2%=X2%
  IF Y<RY1% THEN RY1%=Y
  IF Y>RY2% THEN RY2%=Y

  REM SIMPLE RUNWAY DEPTH SHADING: FAR DARKER, NEAR BRIGHTER.
  RC%=RWC%
  IF N<RW.H%/3 THEN RC%=11
  IF N>=RW.H%/3 THEN IF N<2*RW.H%/3 THEN RC%=12
  IF N>=2*RW.H%/3 THEN RC%=15

  FOR X=X1% TO X2%
    IF Y>=OH%(X) THEN TILE X,Y,RWT%,RC%
  NEXT X

  REM YELLOW EDGE / RANGE MARKERS. SINGLE-CELL DASHES.
  REM V37: SKIP NARROW TIP; SLOW PHASE CHANGE TO AVOID BOUNCE.
  IF N<3 THEN GOTO RWYAPNEXT
  IF X2%-X1%<4 THEN GOTO RWYAPNEXT
  MK%=N+INT(RW.DIST%/4)
  IF MK%<0 THEN MK%=-MK%
  IF MK%-INT(MK%/2)*2<>0 THEN GOTO RWYAPNEXT
  MCLR%=RMK.FG%+RC%*16
  IF Y>=OH%(X1%) THEN TILE X1%,Y,RMK.T%,MCLR%
  IF Y>=OH%(X2%) THEN TILE X2%,Y,RMK.T%,MCLR%
RWYAPNEXT:
NEXT N

IF RX1%<=RX2% AND RY1%<=RY2% THEN RV%=1
RETURN

DRAWRWYNEAR:
REM ON OR PAST THE THRESHOLD: RUNWAY SURFACE EXTENDS FROM THE
REM HORIZON/DISTANT EDGE DOWN TO THE DASH.  TOP ROWS OF% GRASS
REM APPEAR ONLY NEAR THE FAR-END WARNING ZONE.
ENDGRASS%=0
IF RW.DIST%<-50 THEN ENDGRASS%=INT((ABS(RW.DIST%)-50)/4)
IF ENDGRASS%<0 THEN ENDGRASS%=0
IF ENDGRASS%>3 THEN ENDGRASS%=3

RTOP% = OH%(RXC%)+ENDGRASS%
IF RTOP%<VIEW.TOP% THEN RTOP%=VIEW.TOP%
IF RTOP%>VIEW.BOTTOM% THEN RETURN
RW.H% = VIEW.BOTTOM%-RTOP%+1
IF RW.H%<1 THEN RETURN
RW.WBASE%=RW.MAXW%

RY1% = VIEW.BOTTOM%
RY2% = VIEW.TOP%
RX1% = VIEW.RIGHT%
RX2% = VIEW.LEFT%

FOR Y=RTOP% TO VIEW.BOTTOM%
  N=Y-RTOP%+1
  W% = 3 + INT(N*(RW.WBASE%-3)/RW.H%)
  IF W%<3 THEN W%=3
  IF W%>RW.WBASE% THEN W%=RW.WBASE%
  HW% = INT(W%/2)
  X1% = RXC%-HW%
  X2% = RXC%+HW%
  IF X1%<VIEW.LEFT% THEN X1%=VIEW.LEFT%
  IF X2%>VIEW.RIGHT% THEN X2%=VIEW.RIGHT%

  IF X1%<RX1% THEN RX1%=X1%
  IF X2%>RX2% THEN RX2%=X2%
  IF Y<RY1% THEN RY1%=Y
  IF Y>RY2% THEN RY2%=Y

  REM SIMPLE RUNWAY DEPTH SHADING: FAR DARKER, NEAR BRIGHTER.
  RC%=RWC%
  IF N<RW.H%/3 THEN RC%=11
  IF N>=RW.H%/3 THEN IF N<2*RW.H%/3 THEN RC%=12
  IF N>=2*RW.H%/3 THEN RC%=15

  FOR X=X1% TO X2%
    IF Y>=OH%(X)+ENDGRASS% THEN TILE X,Y,RWT%,RC%
  NEXT X

  REM YELLOW EDGE / RANGE MARKERS FOR ON-RUNWAY VIEW.
  REM V37: SINGLE-CELL DASHES; SLOW PHASE CHANGE TO AVOID BOUNCE.
  IF N<3 THEN GOTO RWYNEARNEXT
  IF X2%-X1%<4 THEN GOTO RWYNEARNEXT
  MK%=N+INT(RW.DIST%/4)
  IF MK%<0 THEN MK%=-MK%
  IF MK%-INT(MK%/2)*2<>0 THEN GOTO RWYNEARNEXT
  MCLR%=RMK.FG%+RC%*16
  IF Y>=OH%(X1%)+ENDGRASS% THEN TILE X1%,Y,RMK.T%,MCLR%
  IF Y>=OH%(X2%)+ENDGRASS% THEN TILE X2%,Y,RMK.T%,MCLR%
RWYNEARNEXT:
NEXT Y

IF RX1%<=RX2% AND RY1%<=RY2% THEN RV%=1
RETURN
SMARTERASERUNWAY:
IF ORV%=0 THEN RETURN
FOR Y=ORY1% TO ORY2%
  FOR X=ORX1% TO ORX2%
    KEEP%=0
    IF RV%=1 THEN IF X>=RX1% AND X<=RX2% THEN IF Y>=RY1% AND Y<=RY2% THEN KEEP%=1
    IF KEEP%=0 THEN IF X>=VIEW.LEFT% AND X<=VIEW.RIGHT% THEN GOSUB RESTORECELL
  NEXT X
NEXT Y
RETURN

ERASERUNWAY:
IF RV%=0 THEN RETURN
FOR Y=RY1% TO RY2%
  FOR X=RX1% TO RX2%
    IF X>=VIEW.LEFT% AND X<=VIEW.RIGHT% THEN GOSUB RESTORECELL
  NEXT X
NEXT Y
RV%=0
RETURN

REM === ERASE OBJECTS ===
REM KEPT AS A FALLBACK ROUTINE.
ERASEOBJECTS:
FOR I=1 TO NB%
  IF OV%(I)=0 THEN GOTO ERASEOBJNEXT

  SX%=OS%(I) : TB%=OB%(I) : TTOP%=OT%(I)

  FOR Y=TTOP% TO TB%
    FOR X=SX% TO SX%+OWID%(I)-1
      IF X>=VIEW.LEFT% AND X<=VIEW.RIGHT% THEN GOSUB RESTORECELL
    NEXT X
  NEXT Y

  OV%(I)=0

ERASEOBJNEXT:
NEXT I
RETURN

REM === MOON SKY OBJECT ===
REM APPEARS ONLY ON SOUTHBOUND RUNWAY SETUPS.  POSITION IS WORLD-BEARING
REM BASED SO IT SHIFTS WITH RUDDER/HEADING AND BANKS WITH THE HORIZON.
ERASEMOON:
IF OMV%=0 THEN RETURN
FOR EY=OMY% TO OMY%+1
  FOR EX=OMX% TO OMX%+1
    X=EX:Y=EY
    IF X>=VIEW.LEFT% AND X<=VIEW.RIGHT% THEN IF Y>=VIEW.TOP% AND Y<=VIEW.BOTTOM% THEN GOSUB RESTORECELL
  NEXT EX
NEXT EY
OMV%=0
RETURN

DRAWMOON:
REM V67: MOON IS FIXED IN THE SOUTHERN SKY, NOT RUNWAY-LOCKED.
REM THIS PRESERVES THE OLD SOUTH-APPROACH POSITION: 180+14=194.
MOON.BEAR%=194
MOON.BR%=MOON.BEAR%-HDG%
IF MOON.BR%>180 THEN MOON.BR%=MOON.BR%-360
IF MOON.BR%<-179 THEN MOON.BR%=MOON.BR%+360
REM V62: TREAT MOON AS A SKY POINT ABOVE THE ROLLED HORIZON.
REM USE OPPOSITE X SHIFT SO THE MOON ROTATES WITH THE BANKED SKY.
MOON.SX%=CENTER%+MOON.BR%/2
MOON.H%=6
MOON.X%=INT(MOON.SX%-ROLL%*MOON.H%/10)
MOON.Y%=INT(BASE%+PITCH%-(MOON.SX%-CENTER%)*ROLL%/10-MOON.H%)
IF MOON.X%<VIEW.LEFT% THEN RETURN
IF MOON.X%>VIEW.RIGHT%-1 THEN RETURN
IF MOON.Y%<VIEW.TOP% THEN RETURN
IF MOON.Y%>VIEW.BOTTOM%-1 THEN RETURN
IF MOON.Y%+1>=OH%(MOON.X%) THEN RETURN
IF MOON.Y%+1>=OH%(MOON.X%+1) THEN RETURN
TILE MOON.X%,MOON.Y%,MOON.T1%,MOON.C%
TILE MOON.X%+1,MOON.Y%,MOON.T2%,MOON.C%
TILE MOON.X%,MOON.Y%+1,MOON.T3%,MOON.C%
TILE MOON.X%+1,MOON.Y%+1,MOON.T4%,MOON.C%
OMV%=1:OMX%=MOON.X%:OMY%=MOON.Y%
RETURN
RESTORECELL:
IF Y<OH%(X) THEN TILE X,Y,TS%,CS%
IF Y>=OH%(X) THEN TILE X,Y,TG%,CG%
RETURN

GOSUB DRAWWIND
GOSUB DRAWFLP
GOSUB DRAWRUD
GOSUB DRAWGEAR
GOSUB DRAWRWY
GOSUB DRAWDIST
GOSUB DRAWSTALLWARN
GOSUB DRAWMSG
RETURN

DRAWSPD:
COLOR HC%,0
N=INT(SPD%)
H$=STR$(N):H$=MID$(H$,2)
IF N<100 THEN H$="0"+H$
IF N<10 THEN H$="0"+H$
LOCATE 18,5:PRINT H$;
COLOR 1,0
RETURN

DRAWALT:
COLOR HC%,0
N=INT(ALT%)
H$=STR$(N):H$=MID$(H$,2)
IF N<10000 THEN H$="0"+H$
IF N<1000 THEN H$="0"+H$
IF N<100 THEN H$="0"+H$
IF N<10 THEN H$="0"+H$
LOCATE 18,15:PRINT H$;
COLOR 1,0
RETURN

DRAWFUEL:
COLOR HC%,0
N=INT(FUEL%)
IF N>999 THEN N=999
H$=STR$(N):H$=MID$(H$,2)
IF N<100 THEN H$="0"+H$
IF N<10 THEN H$="0"+H$
LOCATE 18,28:PRINT H$;
COLOR 1,0
RETURN

DRAWVSI:
COLOR HC%,0
N=ABS(INT(VSI%))
IF N>9999 THEN N=9999
H$=STR$(N):H$=MID$(H$,2)
IF N<1000 THEN H$="0"+H$
IF N<100 THEN H$="0"+H$
IF N<10 THEN H$="0"+H$
IF VSI%>=0 THEN H$="+"+H$
IF VSI%<0 THEN H$="-"+H$
LOCATE 19,5:PRINT H$;
COLOR 1,0
RETURN

DRAWTHR:
COLOR HC%,0
LOCATE 19,18:PRINT THR%;
COLOR 1,0
RETURN

DRAWWIND:
GOSUB GETWIND
LOCATE 19,28:PRINT "     ";
LOCATE 19,28:COLOR HC%,0:PRINT WD$;WIND.SPD%;
COLOR 1,0
RETURN

GETWIND:
IF WIND.DIR%=0 THEN WD$="N "
IF WIND.DIR%=1 THEN WD$="NE"
IF WIND.DIR%=2 THEN WD$="E "
IF WIND.DIR%=3 THEN WD$="SE"
IF WIND.DIR%=4 THEN WD$="S "
IF WIND.DIR%=5 THEN WD$="SW"
IF WIND.DIR%=6 THEN WD$="W "
IF WIND.DIR%=7 THEN WD$="NW"
RETURN


DRAWFLP:
COLOR HC%,0
LOCATE 20,6:PRINT FLP%;
COLOR 1,0
RETURN

DRAWRUD:
COLOR HC%,0
N=ABS(INT(RUD%))
H$=STR$(N):H$=MID$(H$,2)
IF N<10 THEN H$="0"+H$
IF RUD%>=0 THEN H$="+"+H$
IF RUD%<0 THEN H$="-"+H$
LOCATE 23,6:PRINT H$;
COLOR 1,0
RETURN

DRAWGEAR:
COLOR 1,0
LOCATE 20,11:PRINT "GEAR        ";
IF GEAR%=1 THEN LOCATE 20,16:COLOR 7,0:PRINT "DOWN";
IF GEAR%=0 THEN LOCATE 20,16:COLOR HC%,0:PRINT "UP  ";
COLOR 1,0
RETURN

DRAWRWY:
COLOR HC%,0
N=INT(RW.BEAR%)
IF N=0 THEN N=360
H$=STR$(N):H$=MID$(H$,2)
IF N<100 THEN H$="0"+H$
IF N<10 THEN H$="0"+H$
LOCATE 22,5:PRINT H$;
COLOR 1,0
RETURN

DRAWDIST:
COLOR HC%,0
N=INT(RW.DIST%)
IF N>=0 THEN GOTO DPOS
N=ABS(N)
IF N>99 THEN N=99
H$=STR$(N):H$=MID$(H$,2)
IF N<10 THEN H$="0"+H$
H$="-"+H$
LOCATE 22,17:PRINT H$;
COLOR 1,0
RETURN
DPOS:
IF N>999 THEN N=999
H$=STR$(N):H$=MID$(H$,2)
IF N<100 THEN H$="0"+H$
IF N<10 THEN H$="0"+H$
LOCATE 22,17:PRINT H$;
COLOR 1,0
RETURN

DRAWSTALLWARN:
IF STALL%=1 THEN COLOR 2,0:LOCATE 17,1:PRINT "STALL";:COLOR 1,0:RETURN
COLOR 1,0
LOCATE 17,1:PRINT "-----";
RETURN

DRAWMSG:
COLOR 1,0
LOCATE 23,24:PRINT "      ";
LOCATE 23,24:COLOR HC%,0
IF MODE%=1 THEN PRINT "LANDED";:COLOR 1,0:RETURN
IF MODE%=2 THEN PRINT "CRASH!";:COLOR 1,0:RETURN
IF MODE%=3 THEN PRINT "MISSED";:COLOR 1,0:RETURN
IF MODE%=4 THEN PRINT "EARLY ";:COLOR 1,0:RETURN
PRINT "------";
COLOR 1,0
RETURN

REM === HEADING DISPLAY ===
DRAWHDG:
DH%=HDG%
IF DH%=0 THEN DH%=360
H$=STR$(DH%)
H$=MID$(H$,2)
IF DH%<100 THEN H$="0"+H$
IF DH%<10 THEN H$="0"+H$
COLOR HC%,1
LOCATE 17,15:PRINT H$;
COLOR 1,0
RETURN
